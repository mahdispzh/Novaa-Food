import { apiClient } from "@/src/shared/api/client";
import { Category } from "../types/category.types";

export type CreateCategoryDto = {
  name: string;
  slug: string;
  image?: string;
  parentId: number | null;
};

export type UpdateCategoryDto = CreateCategoryDto;

export const categoryApi = {
  /**
   * گرفتن لیست دسته‌بندی‌ها (pagination-ready)
   */
  getAll: (page = 1, limit = 10) =>
    apiClient.get<{ data: Category[]; meta: any }>(
      `protected/categories?page=${page}&limit=${limit}`
    ),

  /**
   * گرفتن یک category
   */
  getOne: (id: number) =>
    apiClient.get<Category>(`protected/categories/${id}`),

  /**
   * ساخت category جدید
   */
  create: (data: CreateCategoryDto) =>
    apiClient.post<Category>("protected/categories", data),

  /**
   * آپدیت category
   */
  update: (id: number, data: UpdateCategoryDto) =>
    apiClient.put<Category>(`protected/categories/${id}`, data),

  /**
   * حذف category
   */
  remove: (id: number) =>
    apiClient.delete<{ success: boolean }>(
      `protected/categories/${id}`
    ),
};./**

 * - مدیریت create / update mode
 * - اتصال فرم به API hooks
 * - کنترل open/close
 * - پاس دادن defaultValues برای update

"use client";

import { CategoryForm } from "./category-form";

import { Category } from "../types/category.types";
import {useCategory} from "@/src/app/categories/hook/use-category";
import {useUpdateCategory} from "@/src/app/categories/hook/use-update-category";
import {useCreateCategory} from "@/src/app/categories/hook/use-create-category";

type Props = {
  mode: "create" | "update";
  categoryId?: number;
  onClose: () => void;
};

export function CategoryModal({
                                mode,
                                categoryId,
                                onClose,
                              }: Props) {
  const createMutation = useCreateCategory();
  const updateMutation = useUpdateCategory();

  // فقط در حالت update دیتا می‌گیریم
  const { data: category } = useCategory(
    mode === "update" ? categoryId : undefined
  );

  /**
   * submit handler
   */
  const handleSubmit = (values: any) => {
    if (mode === "create") {
      createMutation.mutate(values, {
        onSuccess: () => onClose(),
      });
      return;
    }

    if (mode === "update" && categoryId) {
      updateMutation.mutate(
        {
          id: categoryId,
          data: values,
        },
        {
          onSuccess: () => onClose(),
        }
      );
    }
  };

  /**
   * default values برای فرم
   */
  const defaultValues =
    mode === "update" && category
      ? {
        name: category.name,
        image: category.image ?? "",
        parentId: category.parent_id,
        slug: category.slug,
      }
      : {
        name: "",
        image: "",
        parentId: null,
        slug: "",
      };

  return (
    <div className="fixed inset-0 flex items-center justify-center bg-black/40">
      <div className="bg-white w-[450px] rounded-sm p-4">
        <CategoryForm
          mode={mode}
          defaultValues={defaultValues}
          onSubmit={handleSubmit}
        />

        <button
          onClick={onClose}
          className="mt-3 text-red-500 w-full"
        >
          بستن
        </button>
      </div>
    </div>
  );
}./**

import { useMutation, useQueryClient } from "@tanstack/react-query";
import { categoryApi } from "../api/category.api";
import { categoryKeys } from "../lib/category-keys";

export function useCreateCategory() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: categoryApi.create,

    onSuccess: () => {
      queryClient.invalidateQueries({
        queryKey: categoryKeys.all,
      });
    },
  });
}./**

import { useMutation, useQueryClient } from "@tanstack/react-query";
import { categoryApi } from "../api/category.api";
import { categoryKeys } from "../lib/category-keys";

export function useDeleteCategory() {
  const queryClient = useQueryClient();

  return useMutation({
    // mutationFn: categoryApi.remove,
    mutationFn: (id: number) => categoryApi.remove(id),
    onSuccess: () => {
      queryClient.invalidateQueries({
        queryKey: categoryKeys.all,
      });
    },
  });
}./**
 * use-update-category.ts
 * ---------------------------------------------
 * update category + refresh cache
 */

import { useMutation, useQueryClient } from "@tanstack/react-query";
import { categoryApi } from "../api/category.api";
import { categoryKeys } from "../lib/category-keys";

export function useUpdateCategory() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: ({ id, data }: any) =>
      categoryApi.update(id, data),

    onSuccess: (_, variables) => {
      queryClient.invalidateQueries({
        queryKey: categoryKeys.all,
      });

      queryClient.invalidateQueries({
        queryKey: categoryKeys.detail(variables.id),
      });
    },
  });
}./**
